01 Singleton (Local Storage)

  1. المشكلة: كل شاشة بتعمل object تخزين جديد، فاللي اتحفظ في Login مش باين في Home.
  2. قبل: BeforeLocalStorage() بيتعمل كل مرة.
  3. مشاكله: بيانات ضايعة، وتحميل من الـ disk أكتر من مرة.
  4. بعد: LocalStorage.instance (strict)، و PrefsStore.instance + interface (soft، زي NetworkManager اللي بيتعمله inject).
  5. الفرق: نسخة واحدة مشتركة.
  6. ليه أحسن: الـ state متسق. والـ soft version بتسيبك تعمل نسخة جديدة في الـ test.
  7. امتى الأولى أحسن؟ لو الـ object ملوش state ورخيص (زي formatter)، اعمله عادي أو static functions، ومش محتاج Singleton خالص.
01_singleton_local_storage.dart Download 
// Singleton — Local Storage
// Run: dart run 01_singleton_local_storage.dart
//
// Typical Flutter shapes: a SharedPreferences wrapper (CacheHelper) and a
// NetworkManager.instance that is still injectable in tests.

// ---------------------------------------------------------------------------
// BEFORE: every screen creates its own storage object.
// ---------------------------------------------------------------------------
class BeforeLocalStorage {
  BeforeLocalStorage() {
    print('BeforeLocalStorage created (loading prefs from disk...)');
  }

  final Map<String, String> _data = {};

  void save(String key, String value) => _data[key] = value;
  String? read(String key) => _data[key];
}

void beforeDemo() {
  BeforeLocalStorage().save('token', 'abc'); // login screen
  final token = BeforeLocalStorage().read('token'); // home screen
  print('Before → token on home screen: $token'); // null! different object
}

// ---------------------------------------------------------------------------
// AFTER (A): strict Singleton — private constructor, one static instance.
// ---------------------------------------------------------------------------
class LocalStorage {
  LocalStorage._() {
    print('LocalStorage created once');
  }

  static final LocalStorage instance = LocalStorage._();

  final Map<String, String> _data = {};

  void save(String key, String value) => _data[key] = value;
  String? read(String key) => _data[key];
}

// ---------------------------------------------------------------------------
// AFTER (B): "soft singleton" — public constructor (tests can make a fresh
// one) + a shared default instance that is handed out through DI.
// This is a common shape for a NetworkManager.
// ---------------------------------------------------------------------------
abstract interface class KeyValueStore {
  void save(String key, String value);
  String? read(String key);
}

class PrefsStore implements KeyValueStore {
  PrefsStore();

  static final PrefsStore instance = PrefsStore();

  final Map<String, String> _data = {};

  @override
  void save(String key, String value) => _data[key] = value;

  @override
  String? read(String key) => _data[key];
}

class SessionCubitLike {
  SessionCubitLike(this._store); // depends on the interface, not on .instance

  final KeyValueStore _store;

  bool get isLoggedIn => _store.read('token') != null;
}

void main() {
  beforeDemo();

  LocalStorage.instance.save('token', 'abc');
  print('After  → token on home screen: ${LocalStorage.instance.read('token')}');
  print('Same object? ${identical(LocalStorage.instance, LocalStorage.instance)}');

  // Production: pass the shared instance.
  final prod = SessionCubitLike(PrefsStore.instance);
  // Test: pass a fresh, isolated store — no shared global state between tests.
  final test = SessionCubitLike(PrefsStore());
  PrefsStore.instance.save('token', 'abc');
  print('prod logged in: ${prod.isLoggedIn}, test logged in: ${test.isLoggedIn}');
}

02 Factory (Notifications)

  1. المشكلة: كل شاشة بتبني الإشعار بإيدها.
  2. قبل: maps مكتوبة يدوي في أكتر من مكان.
  3. مشاكله: typo في الـ route (/attendence)، ومفيش مكان واحد تصلح فيه.
  4. بعد: AppNotification.forKind(kind) و fromMap(map).
  5. الفرق: الـ caller بيقول "عايز إيه" بس.
  6. ليه أحسن: مكان واحد للتعديل + fallback آمن لو جالك نوع مش معروف.
  7. امتى الأولى أحسن؟ لو object واحد بيتعمل في مكان واحد، الـ constructor العادي كفاية.
02_factory_notifications.dart Download 
// Factory (Simple Factory + Dart factory constructor) — Notifications
// Run: dart run 02_factory_notifications.dart
//
// Typical Flutter shapes: NotificationTemplate.forKind(kind) and
// ServerFailure.fromCode(code).

// ---------------------------------------------------------------------------
// BEFORE: every caller builds the notification by hand.
// ---------------------------------------------------------------------------
void beforeDemo() {
  // In the leave-approval screen:
  final a = {'title': 'Leave approved', 'route': '/leave', 'icon': 'check'};
  // In the attendance screen (someone typed it slightly differently):
  final b = {'title': 'Attendance Missing', 'route': '/attendence', 'icon': 'warn'};
  print('Before → $a | $b   (typo in route, no single place to fix)');
}

// ---------------------------------------------------------------------------
// AFTER: one place decides how each kind is built.
// ---------------------------------------------------------------------------
enum NotificationKind { leaveApproved, attendanceMissing, payslipReady }

class AppNotification {
  const AppNotification._({
    required this.title,
    required this.route,
    required this.icon,
  });

  final String title;
  final String route;
  final String icon;

  /// Factory constructor: callers only say *what* they want.
  factory AppNotification.forKind(NotificationKind kind, {String? name}) {
    return switch (kind) {
      NotificationKind.leaveApproved => AppNotification._(
          title: 'Leave approved${name == null ? '' : ' for $name'}',
          route: '/leave',
          icon: 'check',
        ),
      NotificationKind.attendanceMissing => const AppNotification._(
          title: 'Attendance missing',
          route: '/attendance',
          icon: 'warn',
        ),
      NotificationKind.payslipReady => const AppNotification._(
          title: 'Your payslip is ready',
          route: '/payslip',
          icon: 'money',
        ),
    };
  }

  /// Factory constructor that parses data coming from Firestore / FCM.
  factory AppNotification.fromMap(Map<String, dynamic> map) {
    final kind = NotificationKind.values.firstWhere(
      (k) => k.name == map['kind'],
      orElse: () => NotificationKind.attendanceMissing,
    );
    return AppNotification.forKind(kind, name: map['name'] as String?);
  }

  @override
  String toString() => '[$icon] $title → $route';
}

void main() {
  beforeDemo();
  print(AppNotification.forKind(NotificationKind.leaveApproved, name: 'Ahmed'));
  print(AppNotification.fromMap({'kind': 'payslipReady'}));
  print(AppNotification.fromMap({'kind': 'unknown'})); // safe fallback
}

03 Repository (Employee Profile)

  1. المشكلة: الـ Cubit عارف اسم الـ collection وأسماء الـ fields.
  2. قبل: BeforeProfileCubit بيقرا raw['full_name'].
  3. مشاكله: تغيير الـ backend أو اسم field = تعديل في كل Cubit، ومفيش caching، وصعب تعمله test.
  4. بعد: EmployeeRemoteDataSource (raw I/O) ← EmployeeRepository (interface) ← FirestoreEmployeeRepository + InMemoryEmployeeRepository.
  5. الفرق: الـ Cubit بيتكلم بلغة الـ app (Employee) مش بلغة Firestore.
  6. ليه أحسن: تقدر تبدّل الـ repo كله في الـ test، وتضيف cache في مكان واحد.
  7. امتى الأولى أحسن؟ prototype صغير أو شاشة بتقرا document واحد ومش هتتغير. وكتير من الـ apps بتكتفي بـ "Service" بدل Repository، وده مقبول في app بـ backend واحد.
03_repository_employee_profile.dart Download 
// Repository (+ Data Source) — Employee Profile
// Run: dart run 03_repository_employee_profile.dart
//
// A feature repository that combines data sources, placed behind an
// abstract interface so Cubits can be tested with fakes.

// Pretend this is FirebaseFirestore.
class FakeDb {
  final Map<String, Map<String, dynamic>> employees = {
    'e1': {'full_name': 'Sara Ali', 'dept': 'HR', 'salary': 9000},
  };
}

// ---------------------------------------------------------------------------
// BEFORE: the "cubit" talks to the DB, parses maps and applies rules itself.
// ---------------------------------------------------------------------------
class BeforeProfileCubit {
  BeforeProfileCubit(this._db);

  final FakeDb _db;

  String load(String id) {
    final raw = _db.employees[id]; // knows the collection
    if (raw == null) return 'not found';
    final name = raw['full_name'] as String; // knows the field names
    return 'Hello $name';
  }
}

// ---------------------------------------------------------------------------
// AFTER: Model ← DataSource (raw I/O) ← Repository (app-facing API).
// ---------------------------------------------------------------------------
class Employee {
  const Employee({required this.id, required this.name, required this.department});

  final String id;
  final String name;
  final String department;
}

/// Data source: only knows how to talk to one backend. Returns raw data.
class EmployeeRemoteDataSource {
  EmployeeRemoteDataSource(this._db);

  final FakeDb _db;

  Future<Map<String, dynamic>?> fetch(String id) async => _db.employees[id];
}

/// What the rest of the app depends on.
abstract interface class EmployeeRepository {
  Future<Employee?> getEmployee(String id);
}

class FirestoreEmployeeRepository implements EmployeeRepository {
  FirestoreEmployeeRepository(this._remote);

  final EmployeeRemoteDataSource _remote;
  final Map<String, Employee> _cache = {};

  @override
  Future<Employee?> getEmployee(String id) async {
    final cached = _cache[id];
    if (cached != null) return cached;
    final raw = await _remote.fetch(id);
    if (raw == null) return null;
    final employee = Employee(
      id: id,
      name: raw['full_name'] as String,
      department: raw['dept'] as String,
    );
    return _cache[id] = employee;
  }
}

/// In tests (or a demo build) swap the whole repository.
class InMemoryEmployeeRepository implements EmployeeRepository {
  @override
  Future<Employee?> getEmployee(String id) async =>
      const Employee(id: 'x', name: 'Test User', department: 'QA');
}

class ProfileCubit {
  ProfileCubit(this._repo);

  final EmployeeRepository _repo;

  Future<String> load(String id) async {
    final e = await _repo.getEmployee(id);
    return e == null ? 'not found' : 'Hello ${e.name} (${e.department})';
  }
}

Future<void> main() async {
  final db = FakeDb();
  print('Before → ${BeforeProfileCubit(db).load('e1')}');

  final repo = FirestoreEmployeeRepository(EmployeeRemoteDataSource(db));
  print('After  → ${await ProfileCubit(repo).load('e1')}');
  print('Test   → ${await ProfileCubit(InMemoryEmployeeRepository()).load('e1')}');
}

04 Dependency Injection (Attendance)

  1. المشكلة: "هل الموظف متأخر؟" معتمدة على الوقت الحقيقي.
  2. قبل: BeforeAttendanceService بيعمل SystemClock() جواه.
  3. مشاكله: مستحيل تختبر حالة 08:30 و 09:15 بشكل ثابت.
  4. بعد: AttendanceService(clock: ...)، و buildProductionAttendance() = composition root.
  5. الفرق: الـ dependency بتيجي من برة.
  6. ليه أحسن: testable، وتقدر تغير السلوك من غير ما تعدل الـ class. ده نفس اللي بيحصل لما الـ service ياخد FirebaseFirestore? firestore.
  7. امتى الأولى أحسن؟ value objects و helpers صغيرة ملهاش side effects (زي DateFormat) مش محتاجة injection.
04_dependency_injection_attendance.dart Download 
// Dependency Injection — Attendance check-in
// Run: dart run 04_dependency_injection_attendance.dart
//
// Typical Flutter constructor injection, e.g.
// PostsService({FirebaseFirestore? firestore})
// and a composition root at the top of the widget tree.

abstract interface class Clock {
  DateTime now();
}

class SystemClock implements Clock {
  @override
  DateTime now() => DateTime.now();
}

class FixedClock implements Clock {
  FixedClock(this.value);

  final DateTime value;

  @override
  DateTime now() => value;
}

// ---------------------------------------------------------------------------
// BEFORE: the class creates what it needs. Untestable "is late?" logic.
// ---------------------------------------------------------------------------
class BeforeAttendanceService {
  final SystemClock _clock = SystemClock(); // hard-wired

  bool isLate() => _clock.now().hour >= 9; // result depends on when you run it
}

// ---------------------------------------------------------------------------
// AFTER: dependencies come from outside (constructor injection).
// ---------------------------------------------------------------------------
class AttendanceService {
  AttendanceService({required Clock clock, int startHour = 9})
      : _clock = clock,
        _startHour = startHour;

  final Clock _clock;
  final int _startHour;

  bool isLate() => _clock.now().hour >= _startHour;
}

// Composition root: the ONE place that knows the concrete classes
// (in Flutter this is MultiRepositoryProvider in the root widget).
AttendanceService buildProductionAttendance() =>
    AttendanceService(clock: SystemClock());

void main() {
  print('Before → isLate = ${BeforeAttendanceService().isLate()} (depends on real time)');

  final at0830 = AttendanceService(clock: FixedClock(DateTime(2026, 1, 1, 8, 30)));
  final at0915 = AttendanceService(clock: FixedClock(DateTime(2026, 1, 1, 9, 15)));
  print('Test 08:30 → late? ${at0830.isLate()}'); // false, always
  print('Test 09:15 → late? ${at0915.isLate()}'); // true, always
  print('Prod       → late? ${buildProductionAttendance().isLate()}');
}

05 Service Locator vs DI (Permissions)

  1. المشكلة: الـ ViewModel محتاج PermissionService، يجيبه منين؟
  2. قبل (SL): Locator.get<PermissionService>() جوه الـ class.
  3. مشاكله: الـ dependency مخفية، والـ crash بيحصل وقت التشغيل لو نسيت تسجلها.
  4. بعد (DI): LeaveApprovalViewModelDI(PermissionService).
  5. الفرق: SL = الـ class بيسحب. DI = حد تاني بيدفع.
  6. ليه DI أحسن: الـ dependency باينة في الـ constructor، والـ compiler بيلزمك تبعتها.
  7. امتى SL أحسن؟ على الحدود (Screen.builder، main، أو مكان مفيش فيه BuildContext). الحل الشائع: context.read<T>() في الـ builder وبعدين injection للـ Cubit. ده الحل الوسط الصح.
05_service_locator_vs_di.dart Download 
// Service Locator vs Dependency Injection — User permissions
// Run: dart run 05_service_locator_vs_di.dart
//
// In Flutter: context.read<T>() inside Screen.builder() is a *locate* step,
// then the Cubit receives everything through its constructor (DI).

class PermissionService {
  PermissionService(this.role);

  final String role;

  bool canApproveLeave() => role == 'manager' || role == 'hr';
}

// A tiny locator (what get_it does, in 15 lines).
class Locator {
  static final Map<Type, Object> _registry = {};

  static void register<T extends Object>(T instance) => _registry[T] = instance;

  static T get<T extends Object>() {
    final found = _registry[T];
    if (found == null) throw StateError('$T is not registered');
    return found as T;
  }
}

// ---------------------------------------------------------------------------
// Service Locator style: the class PULLS its dependency. Hidden dependency —
// you only discover it when it crashes at runtime.
// ---------------------------------------------------------------------------
class LeaveApprovalViewModelSL {
  bool get showApproveButton => Locator.get<PermissionService>().canApproveLeave();
}

// ---------------------------------------------------------------------------
// DI style: the dependency is PUSHED in. Visible in the constructor.
// ---------------------------------------------------------------------------
class LeaveApprovalViewModelDI {
  LeaveApprovalViewModelDI(this._permissions);

  final PermissionService _permissions;

  bool get showApproveButton => _permissions.canApproveLeave();
}

void main() {
  try {
    LeaveApprovalViewModelSL().showApproveButton;
  } on StateError catch (e) {
    print('SL before registration → crash at runtime: ${e.message}');
  }

  Locator.register(PermissionService('employee'));
  print('SL  → ${LeaveApprovalViewModelSL().showApproveButton}');

  // DI: you can't even build it without giving the dependency (compile-time).
  print('DI  → ${LeaveApprovalViewModelDI(PermissionService('manager')).showApproveButton}');

  // Common Flutter compromise (what many apps do): locate once at the edge,
  // then inject.
  final vm = LeaveApprovalViewModelDI(Locator.get<PermissionService>());
  print('Locate at the edge + inject → ${vm.showApproveButton}');
}

06 Observer (Attendance board)

  1. المشكلة: الشاشة لازم تعرف أول ما حد يعمل check-in.
  2. قبل: polling بـ Timer.periodic.
  3. مشاكله: شغل على الفاضي، وتأخير، وبطارية.
  4. بعد: StreamController.broadcast() + Cubit بيعمل listen + cancel في close().
  5. الفرق: الـ subject بيبعت (push) بدل ما الـ observer يسأل (pull).
  6. ليه أحسن: تحديث لحظي وأكتر من observer مستقلين.
  7. امتى الأولى أحسن؟ لو المصدر مش بيدعم notifications (REST API قديم)، الـ polling هو الحل الوحيد. حطه جوه repository يطلع Stream.
06_observer_attendance_stream.dart Download 
// Observer — Live attendance board (Stream + a Cubit-like class)
// Run: dart run 06_observer_attendance_stream.dart
//
// Typical Flutter shape: NetworkManager.statusStream (broadcast StreamController)
// → NetworkCubit.listen → BlocBuilder.

import 'dart:async';

// ---------------------------------------------------------------------------
// BEFORE: polling — the screen keeps asking "anything new?".
// ---------------------------------------------------------------------------
class BeforeAttendanceSource {
  final List<String> checkedIn = [];
}

Future<void> beforeDemo() async {
  final source = BeforeAttendanceSource();
  var polls = 0;
  final timer = Timer.periodic(const Duration(milliseconds: 10), (_) {
    polls++; // wasted work even when nothing changed
  });
  source.checkedIn.add('Sara');
  await Future<void>.delayed(const Duration(milliseconds: 55));
  timer.cancel();
  print('Before → polled $polls times to notice 1 change');
}

// ---------------------------------------------------------------------------
// AFTER: the subject pushes changes; observers subscribe/unsubscribe.
// ---------------------------------------------------------------------------
class AttendanceFeed {
  final _controller = StreamController<String>.broadcast(); // many listeners

  Stream<String> get checkIns => _controller.stream;

  void checkIn(String name) => _controller.add(name);

  Future<void> dispose() => _controller.close();
}

/// Minimal Cubit: holds state, notifies listeners on every emit.
class CountCubit {
  CountCubit(AttendanceFeed feed) {
    _sub = feed.checkIns.listen((name) => _emit(_state + 1));
  }

  final _states = StreamController<int>.broadcast();
  late final StreamSubscription<String> _sub;
  int _state = 0;

  Stream<int> get stream => _states.stream;

  void _emit(int next) {
    _state = next;
    _states.add(next);
  }

  Future<void> close() async {
    await _sub.cancel(); // always unsubscribe — do it in close()
    await _states.close();
  }
}

Future<void> main() async {
  await beforeDemo();

  final feed = AttendanceFeed();
  final cubit = CountCubit(feed);

  // Two independent observers (think: BlocBuilder + a BlocListener).
  final ui = cubit.stream.listen((n) => print('UI: $n employees in office'));
  final log = feed.checkIns.listen((name) => print('Log: $name checked in'));

  feed
    ..checkIn('Sara')
    ..checkIn('Omar');
  await Future<void>.delayed(Duration.zero);

  await ui.cancel();
  await log.cancel();
  await cubit.close();
  await feed.dispose();
}

07 Strategy (Check-in methods)

  1. المشكلة: الشركات مختلفة: GPS، QR، Wi-Fi، وبكرة face-ID.
  2. قبل: if/else على اسم الطريقة.
  3. مشاكله: كل طريقة جديدة = تعدل نفس الـ function (بيكسر Open/Closed)، وصعب تختبر كل واحدة لوحدها.
  4. بعد: CheckInStrategy + class لكل طريقة، أو function (Verifier) زي navigation hook بيتبدل حسب الـ platform.
  5. الفرق: الـ algorithm بقى object تقدر تبدله وقت التشغيل.
  6. ليه أحسن: تضيف طريقة جديدة من غير ما تلمس القديم.
  7. امتى الأولى أحسن؟ لو هما 2-3 حالات ثابتة ومش هيزيدوا، switch على enum أوضح وأقصر.
07_strategy_checkin_methods.dart Download 
// Strategy — Attendance check-in methods (GPS / QR / Wi-Fi)
// Run: dart run 07_strategy_checkin_methods.dart
//
// A lighter Dart variant is function-based, e.g. a static
// AppNavigator.webPushNamed hook that the web build replaces.

// ---------------------------------------------------------------------------
// BEFORE: one method with a growing if/else. Every new method = edit here.
// ---------------------------------------------------------------------------
bool beforeVerify(String method, Map<String, Object> input) {
  if (method == 'gps') {
    return (input['distanceMeters'] as int) <= 100;
  } else if (method == 'qr') {
    return input['qr'] == 'OFFICE-2026';
  } else if (method == 'wifi') {
    return input['ssid'] == 'Acme-Office';
  }
  throw ArgumentError('unknown method $method');
}

// ---------------------------------------------------------------------------
// AFTER (A): classic Strategy — an interface + one class per algorithm.
// ---------------------------------------------------------------------------
abstract interface class CheckInStrategy {
  String get name;
  bool verify(Map<String, Object> input);
}

class GpsCheckIn implements CheckInStrategy {
  GpsCheckIn({this.maxMeters = 100});

  final int maxMeters;

  @override
  String get name => 'GPS';

  @override
  bool verify(Map<String, Object> input) =>
      (input['distanceMeters'] as int) <= maxMeters;
}

class QrCheckIn implements CheckInStrategy {
  @override
  String get name => 'QR';

  @override
  bool verify(Map<String, Object> input) => input['qr'] == 'OFFICE-2026';
}

class WifiCheckIn implements CheckInStrategy {
  @override
  String get name => 'Wi-Fi';

  @override
  bool verify(Map<String, Object> input) => input['ssid'] == 'Acme-Office';
}

/// The context: does not know which algorithm it runs.
class CheckInService {
  CheckInService(this._strategy);

  CheckInStrategy _strategy;

  set strategy(CheckInStrategy s) => _strategy = s; // swap at runtime

  String checkIn(Map<String, Object> input) =>
      '${_strategy.name}: ${_strategy.verify(input) ? 'accepted' : 'rejected'}';
}

// ---------------------------------------------------------------------------
// AFTER (B): Dart-idiomatic Strategy — just pass a function.
// ---------------------------------------------------------------------------
typedef Verifier = bool Function(Map<String, Object> input);

String checkInWith(Verifier verify, Map<String, Object> input) =>
    verify(input) ? 'accepted' : 'rejected';

void main() {
  print('Before → ${beforeVerify('gps', {'distanceMeters': 40})}');

  final service = CheckInService(GpsCheckIn());
  print(service.checkIn({'distanceMeters': 40}));
  service.strategy = QrCheckIn(); // e.g. company setting changed
  print(service.checkIn({'qr': 'WRONG'}));
  service.strategy = WifiCheckIn();
  print(service.checkIn({'ssid': 'Acme-Office'}));

  print('Function strategy → ${checkInWith((i) => i['badge'] == 42, {'badge': 42})}');
}

08 Adapter (Legacy HR API)

  1. المشكلة: عندك interface جاهز (LeaveBalanceProvider)، وجالك client قديم (Odoo) بأسماء ووحدات مختلفة.
  2. قبل: الشاشة بتحوّل ساعات لأيام بنفسها.

class FakeFirebaseAuth {
  Future<String> signInWithEmailAndPassword(String email, String password) async {
    if (password.length < 6) throw Exception('[firebase_auth/wrong-password]');
    return 'uid-123';
  }
}

class FakeFcm {
  Future<String> getToken() async => 'fcm-token-xyz';
}

class FakeFirestore {
  final Map<String, Map<String, Object>> docs = {};

  Future<void> set(String path, Map<String, Object> data) async => docs[path] = data;
}

// ---------------------------------------------------------------------------
// BEFORE: the login screen orchestrates 3 SDKs + error codes itself.
// ---------------------------------------------------------------------------
Future<String> beforeLoginScreen(String email, String password) async {
  final auth = FakeFirebaseAuth();
  final fcm = FakeFcm();
  final db = FakeFirestore();
  try {
    final uid = await auth.signInWithEmailAndPassword(email.trim(), password);
    final token = await fcm.getToken();
    await db.set('users/$uid/fcmTokens/$token', {'createdAt': 'now'});
    return 'Before → signed in $uid';
  } catch (e) {
    return e.toString().contains('wrong-password') ? 'Wrong password' : 'Error';
  }
}

// ---------------------------------------------------------------------------
// AFTER: one simple entry point; the UI knows nothing about the SDKs.
// ---------------------------------------------------------------------------
sealed class SignInResult {}

class SignedIn extends SignInResult {
  SignedIn(this.uid);

  final String uid;
}

class SignInFailed extends SignInResult {
  SignInFailed(this.message);

  final String message;
}

class AuthFacade {
  AuthFacade({FakeFirebaseAuth? auth, FakeFcm? fcm, FakeFirestore? db})
      : _auth = auth ?? FakeFirebaseAuth(),
        _fcm = fcm ?? FakeFcm(),
        _db = db ?? FakeFirestore();

  final FakeFirebaseAuth _auth;
  final FakeFcm _fcm;
  final FakeFirestore _db;

  Future<SignInResult> signIn(String email, String password) async {
    try {
      final uid = await _auth.signInWithEmailAndPassword(email.trim(), password);
      await _registerPushToken(uid);
      return SignedIn(uid);
    } catch (e) {
      return SignInFailed(_mapError(e));
    }
  }

  Future<void> _registerPushToken(String uid) async {
    final token = await _fcm.getToken();
    await _db.set('users/$uid/fcmTokens/$token', {'createdAt': 'now'});
  }

  String _mapError(Object e) =>
      e.toString().contains('wrong-password') ? 'Wrong password' : 'Something went wrong';
}

    print(switch (result) {

// Decorator — add caching + logging to a repository without editing it
// Run: dart run 10_decorator_cached_repository.dart
//
// Widget wrappers like
// PushNotificationListener(child: ...) are the Flutter-composition cousin.

abstract interface class DashboardStatsRepository {
  Future<int> activeEmployees(String companyId);
}

class RemoteDashboardStatsRepository implements DashboardStatsRepository {
  int calls = 0;

  @override
  Future<int> activeEmployees(String companyId) async {
    calls++;
    await Future<void>.delayed(const Duration(milliseconds: 20)); // network
    return 42;
  }
}

// ---------------------------------------------------------------------------
// BEFORE: caching + logging pasted inside the remote class (or in each cubit).
// Every new concern makes the class bigger; you can't turn one off.
// ---------------------------------------------------------------------------
class BeforeStatsRepository {
  final Map<String, int> _cache = {};

  Future<int> activeEmployees(String companyId) async {
    print('[log] activeEmployees($companyId)');
    final hit = _cache[companyId];
    if (hit != null) return hit;
    await Future<void>.delayed(const Duration(milliseconds: 20));
    return _cache[companyId] = 42;
  }
}

// ---------------------------------------------------------------------------
// AFTER: each concern is a wrapper with the SAME interface.
// ---------------------------------------------------------------------------
class CachedStatsRepository implements DashboardStatsRepository {
  CachedStatsRepository(this._inner);

  final DashboardStatsRepository _inner;
  final Map<String, int> _cache = {};

  @override
  Future<int> activeEmployees(String companyId) async =>
      _cache[companyId] ??= await _inner.activeEmployees(companyId);
}

class LoggingStatsRepository implements DashboardStatsRepository {
  LoggingStatsRepository(this._inner);

  final DashboardStatsRepository _inner;

  @override
  Future<int> activeEmployees(String companyId) async {
    final sw = Stopwatch()..start();
    final result = await _inner.activeEmployees(companyId);
    print('[log] activeEmployees($companyId) = $result in ${sw.elapsedMilliseconds}ms');
    return result;
  }
}

Future<void> main() async {
  await BeforeStatsRepository().activeEmployees('acme');

  final remote = RemoteDashboardStatsRepository();
  // Stack them in any order; the cubit only sees DashboardStatsRepository.
  final DashboardStatsRepository repo = LoggingStatsRepository(CachedStatsRepository(remote));
  await repo.activeEmployees('acme');
  await repo.activeEmployees('acme'); // served from cache, much faster
  print('Remote was called ${remote.calls} time(s)');
}

// Command — Offline attendance queue (+ undo)
// Run: dart run 11_command_offline_queue.dart
//
// Typical Flutter shape: a NetworkRequestQueue that stores closures and
// replays them on reconnect.

// ---------------------------------------------------------------------------
// BEFORE: the action runs immediately. If offline, it's just lost.
// ---------------------------------------------------------------------------
Future<void> beforeCheckIn(bool online, String employee) async {
  if (!online) {
    print('Before → offline, check-in for $employee LOST');
    return;
  }
  print('Before → sent check-in for $employee');
}

// ---------------------------------------------------------------------------
// AFTER: each action is an object → can be queued, retried, logged, undone.
// ---------------------------------------------------------------------------
abstract interface class AttendanceCommand {
  String get label;
  Future<void> execute();
  Future<void> undo();
}

class FakeAttendanceApi {
  final List<String> records = [];
}

class CheckInCommand implements AttendanceCommand {
  CheckInCommand(this._api, this.employee, this.at);

  final FakeAttendanceApi _api;
  final String employee;
  final DateTime at;

  @override
  String get label => 'check-in $employee @ ${at.hour}:${at.minute.toString().padLeft(2, '0')}';

  @override
  Future<void> execute() async => _api.records.add(label);

  @override
  Future<void> undo() async => _api.records.remove(label);
}

class CommandQueue {
  final List<AttendanceCommand> _pending = [];
  final List<AttendanceCommand> _history = [];

  Future<void> run(AttendanceCommand c, {required bool online}) async {
    if (!online) {
      _pending.add(c);
      print('After  → queued "${c.label}" (${_pending.length} pending)');
      return;
    }
    await c.execute();
    _history.add(c);
  }

  Future<void> flush() async {
    final copy = List.of(_pending);
    _pending.clear();
    for (final c in copy) {
      await c.execute();
      _history.add(c);
      print('After  → replayed "${c.label}"');
    }
  }

  Future<void> undoLast() async {
    if (_history.isEmpty) return;
    final c = _history.removeLast();
    await c.undo();
    print('After  → undid "${c.label}"');
  }
}

Future<void> main() async {
  await beforeCheckIn(false, 'Sara');

  final api = FakeAttendanceApi();
  final queue = CommandQueue();
  await queue.run(CheckInCommand(api, 'Sara', DateTime(2026, 1, 1, 8, 55)), online: false);
  await queue.run(CheckInCommand(api, 'Omar', DateTime(2026, 1, 1, 9, 2)), online: false);
  await queue.flush(); // connection is back
  await queue.undoLast(); // HR fixes a mistake
  print('Saved records: ${api.records}');
}

// State (GoF) vs enum status — Leave request workflow
// Run: dart run 12_state_leave_request.dart
//
// Many apps use enum statuses (e.g. AuthUiStatus { idle, loading, success, error }).
// That is fine for UI status. The GoF State pattern pays off when each state
// has DIFFERENT allowed actions.

// ---------------------------------------------------------------------------
// BEFORE: enum + switch in every method. Rules are scattered.
// ---------------------------------------------------------------------------
enum LeaveStatus { draft, submitted, approved, rejected }

class BeforeLeaveRequest {
  LeaveStatus status = LeaveStatus.draft;

  void submit() {
    if (status != LeaveStatus.draft) throw StateError('Only drafts can be submitted');
    status = LeaveStatus.submitted;
  }

  void approve() {
    if (status != LeaveStatus.submitted) throw StateError('Only submitted can be approved');
    status = LeaveStatus.approved;
  }

  void cancel() {
    // Easy to forget a case: should an approved leave be cancellable?
    if (status == LeaveStatus.rejected) throw StateError('Already rejected');
    status = LeaveStatus.draft;
  }
}

// ---------------------------------------------------------------------------
// AFTER: each state is an object that decides what each action does.
// ---------------------------------------------------------------------------
sealed class LeaveState {
  const LeaveState();

  String get label;
  LeaveState submit() => throw StateError('Cannot submit when $label');
  LeaveState approve() => throw StateError('Cannot approve when $label');
  LeaveState reject() => throw StateError('Cannot reject when $label');
  LeaveState cancel() => throw StateError('Cannot cancel when $label');
}

class Draft extends LeaveState {
  const Draft();

  @override
  String get label => 'Draft';

  @override
  LeaveState submit() => const Submitted();
}

class Submitted extends LeaveState {
  const Submitted();

  @override
  String get label => 'Submitted';

  @override
  LeaveState approve() => const Approved();

  @override
  LeaveState reject() => const Rejected();

  @override
  LeaveState cancel() => const Draft();
}

class Approved extends LeaveState {
  const Approved();

  @override
  String get label => 'Approved';
}

class Rejected extends LeaveState {
  const Rejected();

  @override
  String get label => 'Rejected';
}

class LeaveRequest {
  LeaveState _state = const Draft();

  String get status => _state.label;

  void submit() => _state = _state.submit();
  void approve() => _state = _state.approve();
  void reject() => _state = _state.reject();
  void cancel() => _state = _state.cancel();
}

void main() {
  final before = BeforeLeaveRequest()
    ..submit()
    ..approve()
    ..cancel(); // bug: approved leave silently went back to draft
  print('Before → ${before.status.name} (should not be allowed)');

  final leave = LeaveRequest()
    ..submit()
    ..approve();
  print('After  → ${leave.status}');
  try {
    leave.cancel();
  } on StateError catch (e) {
    print('After  → ${e.message}');
  }
}

// Builder — Attendance report request with many optional parts
// Run: dart run 13_builder_report_request.dart
//
// Note: a screen's `static Widget builder(BuildContext)` on screens is a static
// FACTORY, not this pattern. BlocBuilder/LayoutBuilder are not GoF Builder either.

// ---------------------------------------------------------------------------
// BEFORE: a constructor with many positional/optional args.
// ---------------------------------------------------------------------------
class BeforeReportRequest {
  BeforeReportRequest(this.companyId, this.from, this.to, this.dept, this.onlyLate,
      this.includeWeekends, this.format);

  final String companyId;
  final DateTime from;
  final DateTime to;
  final String? dept;
  final bool onlyLate;
  final bool includeWeekends;
  final String format;
}

// AFTER: build step by step, validate once in build().
// ---------------------------------------------------------------------------
class ReportRequest {
  const ReportRequest._(this.query);

  final Map<String, String> query;

  @override
  String toString() => '/reports/attendance?${query.entries.map((e) => '${e.key}=${e.value}').join('&')}';
}

class ReportRequestBuilder {
  ReportRequestBuilder(this._companyId);

  final String _companyId;
  DateTime? _from;
  DateTime? _to;
  String? _dept;
  bool _onlyLate = false;
  String _format = 'pdf';

  ReportRequestBuilder between(DateTime from, DateTime to) {
    _from = from;
    _to = to;
    return this;
  }

  ReportRequestBuilder department(String dept) {
    _dept = dept;
    return this;
  }

  ReportRequestBuilder onlyLateArrivals() {
    _onlyLate = true;
    return this;
  }

  ReportRequestBuilder asExcel() {
    _format = 'xlsx';
    return this;
  }

  ReportRequest build() {
    final from = _from, to = _to;
    if (from == null || to == null) throw StateError('Date range is required');
    if (to.isBefore(from)) throw StateError('"to" is before "from"');
    String d(DateTime x) => x.toIso8601String().substring(0, 10);
    return ReportRequest._({
      'company': _companyId,
      'from': d(from),
      'to': d(to),
      if (_dept != null) 'dept': _dept!,
      if (_onlyLate) 'late': 'true',
      'format': _format,
    });
  }
}

14 Result (Error handling)

  1. المشكلة: الـ function ممكن تفشل، بس الـ signature مش بيقول كده.
  2. قبل: Future<double> بيرمي exception.
  3. مشاكله: لو نسيت try/catch مرة = crash.
  4. بعد: Future<Result<double>> + sealed + switch exhaustive.
  5. الفرق: الفشل بقى قيمة عادية الـ compiler بيلزمك تتعامل معاها.
  6. ليه أحسن: لو عندك Failure classes، الـ Result هو الخطوة اللي جاية.
  7. امتى الأولى أحسن؟ الأخطاء النادرة جداً (bugs، StateError) خليها exceptions. الـ Result للأخطاء المتوقعة (network، permission).

// Result type — errors as values (Flutter docs "Result" pattern)
// Run: dart run 14_result_error_handling.dart
//
// Common today: services throw → Cubit catches → ErrorHandler.handle(e)
// → Failure.
// The Failure classes exist; a Result wrapper is the next step.

class Failure {
  const Failure(this.message);

  final String message;
}

class FakeApi {
  Future<double> fetchSalary(String id) async {
    if (id == 'bad') throw Exception('permission-denied');
    return 9000;
  }
}

// ---------------------------------------------------------------------------
// BEFORE: the signature says "returns double" but it can also throw.
// Forget the try/catch once → crash.
// ---------------------------------------------------------------------------
Future<String> beforeCubit(FakeApi api, String id) async {
  final salary = await api.fetchSalary(id); // nothing forces you to catch
  return 'Before → $salary';
}

// ---------------------------------------------------------------------------
// AFTER: the signature says success OR failure. The compiler makes you
// handle both (sealed + exhaustive switch).
// ---------------------------------------------------------------------------
sealed class Result<T> {
  const Result();
}

class Ok<T> extends Result<T> {
  const Ok(this.value);

  final T value;
}

class Err<T> extends Result<T> {
  const Err(this.failure);

  final Failure failure;
}

class SalaryRepository {
  SalaryRepository(this._api);

  final FakeApi _api;

  Future<Result<double>> salary(String id) async {
    try {
      return Ok(await _api.fetchSalary(id));
    } catch (e) {
      final denied = e.toString().contains('permission-denied');
      return Err(Failure(denied ? 'You are not allowed to see this salary' : 'Try again'));
    }
  }
}

Future<void> main() async {
  try {
    print(await beforeCubit(FakeApi(), 'bad'));
  } catch (e) {
    print('Before → uncaught: $e');
  }

  final repo = SalaryRepository(FakeApi());
  for (final id in ['e1', 'bad']) {
    final text = switch (await repo.salary(id)) {
      Ok(:final value) => 'After  → salary $value',
      Err(:final failure) => 'After  → ${failure.message}',
    };
    print(text);
  }
}
